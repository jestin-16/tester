PROBLEM:
Build a Decision Tree to classify Bank Loan Eligibility (Eligible vs Not Eligible).
  - Calculate Target Entropy H(S).
  - Calculate Information Gain for all attributes: Income, Employment, Credit_Rating, Existing_Loan.
  - Identify Root Node (Attribute with Highest Information Gain).
  - Construct Recursive Decision Tree.
  - Extract IF-THEN Decision Rules and predict for a new applicant.

--------------------------------------------------------------------------------
1. PYTHON CODE (COPY & RUN DIRECTLY)
--------------------------------------------------------------------------------
import math
import pandas as pd

# Step 1: Create 15-Customer Loan Dataset
data = {
    "Income":        ["High", "High", "High", "Medium", "Medium", "Medium", "Low", "Low", "Low", "High", "Medium", "Low", "High", "Medium", "Low"],
    "Employment":    ["Permanent", "Permanent", "Temporary", "Permanent", "Temporary", "Unemployed", "Permanent", "Temporary", "Unemployed", "Permanent", "Unemployed", "Permanent", "Temporary", "Permanent", "Unemployed"],
    "Credit_Rating": ["Good", "Good", "Poor", "Good", "Average", "Poor", "Good", "Average", "Poor", "Average", "Good", "Poor", "Good", "Average", "Good"],
    "Existing_Loan": ["No", "Yes", "No", "No", "Yes", "Yes", "No", "No", "Yes", "Yes", "No", "Yes", "No", "No", "No"],
    "Loan_Eligibility": ["Eligible", "Eligible", "Not Eligible", "Eligible", "Eligible", "Not Eligible", "Eligible", "Not Eligible", "Not Eligible", "Eligible", "Not Eligible", "Not Eligible", "Eligible", "Eligible", "Not Eligible"]
}

df = pd.DataFrame(data)
features = ["Income", "Employment", "Credit_Rating", "Existing_Loan"]
target = "Loan_Eligibility"

# Step 2: Entropy Formula
def entropy(series):
    counts = series.value_counts()
    total = len(series)
    if total == 0: return 0
    return -sum((c / total) * math.log2(c / total) for c in counts if c > 0)

target_entropy = entropy(df[target])

# Step 3: Information Gain Formula
def information_gain(df, attribute, target):
    parent_entropy = entropy(df[target])
    total = len(df)
    weighted_entropy = sum((len(sub) / total) * entropy(sub[target]) for _, sub in df.groupby(attribute))
    return parent_entropy - weighted_entropy

gain_dict = {feat: information_gain(df, feat, target) for feat in features}
root_node = max(gain_dict, key=gain_dict.get)

# Step 4: Recursive Decision Tree Builder
def build_tree(df, features, target):
    if len(df[target].unique()) == 1:
        return df[target].iloc[0]
    if len(features) == 0:
        return df[target].mode()[0]
    
    gains = {f: information_gain(df, f, target) for f in features}
    best_feat = max(gains, key=gains.get)
    
    tree = {best_feat: {}}
    rem_features = [f for f in features if f != best_feat]
    
    for val, subset in df.groupby(best_feat):
        tree[best_feat][val] = build_tree(subset, rem_features, target)
        
    return tree

tree = build_tree(df, features, target)

# Step 5: Extract Rules & Predict
def print_rules(tree, path=""):
    if not isinstance(tree, dict):
        print(f"IF {path} THEN Eligibility = {tree}")
        return
    for feat, branches in tree.items():
        for val, subtree in branches.items():
            new_path = f"({feat} == '{val}')" if not path else f"{path} AND ({feat} == '{val}')"
            print_rules(subtree, new_path)

def predict(tree, sample):
    if not isinstance(tree, dict):
        return tree
    feat = next(iter(tree))
    val = sample.get(feat)
    if val in tree[feat]:
        return predict(tree[feat][val], sample)
    return "Not Eligible"

print("=== 1. ENTROPY & INFORMATION GAIN ===")
print(f"Dataset Target Entropy H(S) = {target_entropy:.4f}\n")
for f, g in gain_dict.items():
    print(f"   Information Gain(S, {f:<14}) = {g:.4f}")
print(f"\n[+] Root Node Feature Selected: '{root_node}' (Highest Gain)")

print("\n=== 2. EXTRACTED IF-THEN RULES ===")
print_rules(tree)

new_applicant = {"Income": "Medium", "Employment": "Permanent", "Credit_Rating": "Good", "Existing_Loan": "No"}
pred = predict(tree, new_applicant)
print(f"\n=== 3. PREDICTION FOR NEW APPLICANT ===")
print(f"Applicant: {new_applicant}")
print(f"Prediction: {pred}")
