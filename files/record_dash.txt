import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# Step 1: Create Retail Dataset (100 Records)
np.random.seed(42)
n = 100
months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"]
regions = ["North", "South", "East", "West"]

data = {
    "StoreID": [f"STR{i:03d}" for i in range(1, n + 1)],
    "Month": np.random.choice(months, size=n),
    "Footfall": np.random.randint(1000, 8000, size=n),
    "Sales_Revenue": np.random.normal(50000, 12000, size=n).round(2),
    "Ad_Spend": np.random.uniform(2000, 15000, size=n).round(2),
    "Customer_Satisfaction": np.random.uniform(3.0, 5.0, size=n).round(1),
    "Region": np.random.choice(regions, size=n)
}

df = pd.DataFrame(data)

# Step 2: Create Unified 6-Plot Dashboard (3x2 Subplots)
fig, axes = plt.subplots(3, 2, figsize=(14, 12))
fig.suptitle("Retail Operations & Financial Analytics Dashboard", fontsize=16, fontweight="bold", y=0.98)

# 1. Line Chart: Monthly Sales Revenue Trend
monthly_sales = df.groupby("Month")["Sales_Revenue"].mean().reindex(months)
axes[0, 0].plot(months, monthly_sales.values, marker="o", color="royalblue", linewidth=2.5, linestyle="-")
axes[0, 0].set_title("1. Average Monthly Sales Revenue Trend", fontweight="bold")
axes[0, 0].set_xlabel("Month")
axes[0, 0].set_ylabel("Sales ($)")
axes[0, 0].grid(True, linestyle="--", alpha=0.6)
axes[0, 0].tick_params(axis="x", rotation=45)

# 2. Bar Chart: Total Footfall per Region
region_footfall = df.groupby("Region")["Footfall"].sum()
bars = axes[0, 1].bar(region_footfall.index, region_footfall.values, color=["#ff9999", "#66b3ff", "#99ff99", "#ffcc99"], edgecolor="black")
axes[0, 1].set_title("2. Total Customer Footfall by Region", fontweight="bold")
axes[0, 1].set_xlabel("Region")
axes[0, 1].set_ylabel("Total Footfall")
axes[0, 1].bar_label(bars, padding=3)
axes[0, 1].grid(axis="y", linestyle=":", alpha=0.6)

# 3. Histogram: Customer Footfall Distribution
axes[1, 0].hist(df["Footfall"], bins=10, color="teal", edgecolor="black", alpha=0.75)
axes[1, 0].set_title("3. Distribution of Store Footfall", fontweight="bold")
axes[1, 0].set_xlabel("Footfall Intervals")
axes[1, 0].set_ylabel("Frequency (Number of Stores)")
axes[1, 0].grid(axis="y", linestyle="--", alpha=0.6)

# 4. Scatter Plot: Ad Spend vs Sales Revenue
scatter = axes[1, 1].scatter(df["Ad_Spend"], df["Sales_Revenue"], c=df["Customer_Satisfaction"], cmap="plasma", s=80, alpha=0.85, edgecolors="black")
axes[1, 1].set_title("4. Ad Spend vs Sales Revenue (Color = Satisfaction)", fontweight="bold")
axes[1, 1].set_xlabel("Advertising Spend ($)")
axes[1, 1].set_ylabel("Sales Revenue ($)")
axes[1, 1].grid(True, linestyle="--", alpha=0.6)
fig.colorbar(scatter, ax=axes[1, 1], label="Satisfaction (1-5)")

# 5. Box Plot: Sales Revenue Across Regions (Outlier Detection)
sns.boxplot(data=df, x="Region", y="Sales_Revenue", ax=axes[2, 0], palette="Set3")
axes[2, 0].set_title("5. Regional Sales Distribution & Outliers", fontweight="bold")
axes[2, 0].set_xlabel("Region")
axes[2, 0].set_ylabel("Sales Revenue ($)")
axes[2, 0].grid(axis="y", linestyle="--", alpha=0.6)

# 6. Heatmap: Inter-Feature Correlation Matrix
num_cols = df.select_dtypes(include="number")
corr = num_cols.corr()
sns.heatmap(corr, annot=True, cmap="coolwarm", fmt=".2f", ax=axes[2, 1], cbar=True, linewidths=0.5)
axes[2, 1].set_title("6. Inter-Metric Correlation Heatmap", fontweight="bold")

# Adjust layout and Save Image
plt.tight_layout()
plt.savefig("Retail_Analytics_Dashboard.png", dpi=300, bbox_inches="tight")
plt.show()
