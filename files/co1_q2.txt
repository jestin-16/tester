PROBLEM:
Load an Employee Performance Dataset (30 employees, 5 evaluation criteria):
  - Perform: Mean, Median, Mode, Max, Min, Sum, Average, Std Dev, Variance.
  - Prepare Department-wise Performance Report.
  - Identify the Best-Performing Department based on statistical results.

--------------------------------------------------------------------------------
1. PYTHON CODE (COPY & RUN DIRECTLY)
--------------------------------------------------------------------------------
import numpy as np
import pandas as pd

# Step 1: Create 30 Employees Dataset
np.random.seed(101)
criteria = ["Quality", "Productivity", "Teamwork", "Innovation", "Punctuality"]
departments = ["IT", "HR", "Sales", "Finance", "Marketing"]

data = {
    "EmpID": [f"EMP{i:03d}" for i in range(1, 31)],
    "Name": [f"Employee_{i}" for i in range(1, 31)],
    "Department": np.random.choice(departments, size=30),
    "Experience": np.random.randint(1, 15, size=30)
}

# Score each criterion from 50 to 100
for c in criteria:
    data[c] = np.random.randint(50, 100, size=30)

df = pd.DataFrame(data)

# Step 2: Criteria-wise Statistical Summary
stats_df = pd.DataFrame({
    "Mean": df[criteria].mean().round(2),
    "Median": df[criteria].median().round(2),
    "Mode": df[criteria].mode().iloc[0],
    "Maximum": df[criteria].max(),
    "Minimum": df[criteria].min(),
    "Sum": df[criteria].sum(),
    "Std_Dev": df[criteria].std().round(2),
    "Variance": df[criteria].var().round(2)
})

# Step 3: Employee-wise Total and Average Scores
df["Total_Score"] = df[criteria].sum(axis=1)
df["Average_Score"] = (df["Total_Score"] / len(criteria)).round(2)

# Step 4: Department-wise Performance Report
dept_report = df.groupby("Department").agg(
    Employee_Count=("EmpID", "count"),
    Avg_Experience=("Experience", "mean"),
    Avg_Performance=("Average_Score", "mean"),
    Min_Performance=("Average_Score", "min"),
    Max_Performance=("Average_Score", "max"),
    Performance_Std=("Average_Score", "std")
).reset_index().round(2)

# Step 5: Identify Best-Performing Department
best_dept = dept_report.loc[dept_report["Avg_Performance"].idxmax()]

# Step 6: Print Results
print("=== 1. EMPLOYEE DATASET (FIRST 5 RECORDS) ===")
print(df[["EmpID", "Department", "Experience"] + criteria].head().to_string(index=False))

print("\n=== 2. CRITERIA-WISE STATISTICAL SUMMARY ===")
print(stats_df.to_string())

print("\n=== 3. DEPARTMENT-WISE PERFORMANCE REPORT ===")
print(dept_report.to_string(index=False))

print("\n=== 4. BEST PERFORMING DEPARTMENT ===")
print(f"Department Name:        {best_dept['Department']}")
print(f"Average Score:          {best_dept['Avg_Performance']} / 100")
print(f"Total Employees:        {best_dept['Employee_Count']}")
print(f"Avg Years Experience:   {best_dept['Avg_Experience']} years")
