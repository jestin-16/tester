================================================================================
k-NN FROM SCRATCH -- HIGH PRIORITY
Algorithm: preprocess -> calculate distance -> sort -> choose k -> majority vote
================================================================================

import numpy as np

def euclidean(a, b):
    return np.sqrt(np.sum((a - b) ** 2))

def manhattan(a, b):
    return np.sum(np.abs(a - b))

def minkowski(a, b, p=3):
    return np.sum(np.abs(a - b) ** p) ** (1 / p)

def chebyshev(a, b):
    return np.max(np.abs(a - b))

def knn_predict(X_train, y_train, x_test, k=5):
    distances = []
    for i in range(len(X_train)):
        d = euclidean(X_train[i], x_test)
        distances.append((d, y_train[i]))
    distances.sort(key=lambda x: x[0])
    neighbours = distances[:k]
    labels = [label for _, label in neighbours]
    return max(set(labels), key=labels.count)

# Weighted k-NN formula:
# weight = 1 / (distance + 0.0001)
