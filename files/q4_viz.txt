EXERCISE 3: BAR CHARTS (HORIZONTAL & GROUPED WITH DATA LABELS)
--------------------------------------------------------------------------------
import numpy as np

depts = ["CSE", "ECE", "MECH", "CIVIL", "IT"]
boys = [60, 45, 75, 40, 50]
girls = [55, 50, 15, 20, 45]

# Grouped Bar Chart
x = np.arange(len(depts))
width = 0.35

fig, ax = plt.subplots(figsize=(8, 5))
rects1 = ax.bar(x - width/2, boys, width, label="Boys", color="royalblue")
rects2 = ax.bar(x + width/2, girls, width, label="Girls", color="lightcoral")

# Add data labels
ax.bar_label(rects1, padding=3)
ax.bar_label(rects2, padding=3)

ax.set_xticks(x)
ax.set_xticklabels(depts)
ax.set_title("Department Student Strength by Gender")
ax.set_ylabel("Number of Students")
ax.legend()
plt.show()

# Horizontal Bar Chart
total_students = [b + g for b, g in zip(boys, girls)]
plt.figure(figsize=(7, 4))
bars = plt.barh(depts, total_students, color="teal")
plt.bar_label(bars, padding=3)
plt.title("Total Students per Department (Horizontal Bar)")
plt.xlabel("Total Count")
plt.show()

EXERCISE 6: SCATTER PLOT (BMI SIZE & COLOR WITH TRANSPARENCY)
--------------------------------------------------------------------------------
np.random.seed(42)
heights = np.random.uniform(150, 190, 30)
weights = np.random.uniform(50, 100, 30)
bmi = weights / ((heights / 100) ** 2)

plt.figure(figsize=(7, 5))
scatter = plt.scatter(heights, weights, c=bmi, s=bmi*10, cmap="viridis", alpha=0.75, edgecolors="black")
plt.colorbar(scatter, label="BMI")
plt.title("Student Height vs Weight (Size & Color = BMI)")
plt.xlabel("Height (cm)")
plt.ylabel("Weight (kg)")
plt.grid(True)
plt.show()
# Observation: Positive correlation exists; weight generally increases with height.

EXERCISE 7: BOX PLOT (COMPARING 2 CLASSES & OUTLIER DETECTION)
--------------------------------------------------------------------------------
np.random.seed(42)
class_A = np.concatenate([np.random.normal(72, 10, 45), [20, 25, 99]])  # 2 low outliers
class_B = np.random.normal(68, 8, 48)

plt.figure(figsize=(7, 5))
plt.boxplot([class_A, class_B], tick_labels=["Class A", "Class B"], patch_artist=True)
plt.title("Performance Comparison with Outliers")
plt.ylabel("Marks")
plt.grid(True)
plt.show()

# Box Plot Terminology:
#   Median (Q2): Center line of the box (50th percentile).
#   Quartiles (Q1, Q3): Lower (25th) and Upper (75th) boundaries of the box. IQR = Q3 - Q1.
#   Whiskers: Lines extending to Q1 - 1.5*IQR and Q3 + 1.5*IQR.
#   Outliers: Data points plotted individually beyond the whiskers.

EXERCISE 8: SEABORN VISUALIZATIONS (COUNT, BAR, BOX, VIOLIN, STRIP)
--------------------------------------------------------------------------------
import seaborn as sns
import pandas as pd

np.random.seed(42)
df_tips = pd.DataFrame({
    "day": np.random.choice(["Thur", "Fri", "Sat", "Sun"], 100),
    "total_bill": np.random.uniform(10, 50, 100).round(2),
    "sex": np.random.choice(["Male", "Female"], 100)
})

fig, axes = plt.subplots(2, 3, figsize=(14, 8))
sns.countplot(data=df_tips, x="day", ax=axes[0, 0]); axes[0, 0].set_title("1. Count Plot")
sns.barplot(data=df_tips, x="day", y="total_bill", ax=axes[0, 1]); axes[0, 1].set_title("2. Bar Plot (Mean Bill)")
sns.boxplot(data=df_tips, x="day", y="total_bill", ax=axes[0, 2]); axes[0, 2].set_title("3. Box Plot")
sns.violinplot(data=df_tips, x="day", y="total_bill", ax=axes[1, 0]); axes[1, 0].set_title("4. Violin Plot")
sns.stripplot(data=df_tips, x="day", y="total_bill", jitter=True, ax=axes[1, 1]); axes[1, 1].set_title("5. Strip Plot")
axes[1, 2].axis("off")
plt.tight_layout()
plt.show()

EXERCISE 9: CORRELATION HEATMAP (HIGHEST PAIR IDENTIFICATION)
--------------------------------------------------------------------------------
np.random.seed(42)
df_corr = pd.DataFrame({
    "Study_Hours": np.random.randint(1, 10, 50),
    "Attendance": np.random.randint(60, 100, 50),
    "Internal_Marks": np.random.randint(15, 30, 50),
    "Final_Score": np.random.randint(50, 100, 50)
})

plt.figure(figsize=(7, 5))
corr_matrix = df_corr.corr()
sns.heatmap(corr_matrix, annot=True, cmap="coolwarm", fmt=".2f", linewidths=0.5)
plt.title("Correlation Heatmap of Student Academic Metrics")
plt.show()

# Highest Correlation Pair logic:
pairs = corr_matrix.abs().unstack()
pairs = pairs[pairs < 1.0]
print("Highest Correlation Pair:", pairs.idxmax(), "with r =", pairs.max())

EXERCISE 13: REAL DATASET VISUALIZATION (MINIMUM 50 RECORDS)
--------------------------------------------------------------------------------
np.random.seed(42)
df_real = pd.DataFrame({
    "Age": np.random.randint(18, 60, 60),
    "Salary": np.random.randint(25000, 120000, 60),
    "Dept": np.random.choice(["HR", "IT", "Sales", "Finance"], 60),
    "Experience": np.random.randint(1, 20, 60)
})

# 1. Display first 5 records & Identify columns
print("First 5 records:\n", df_real.head())
print("Numerical Columns:   ", list(df_real.select_dtypes(include="number").columns))
print("Categorical Columns: ", list(df_real.select_dtypes(include="object").columns))

# 2. Plots
fig, axes = plt.subplots(2, 2, figsize=(11, 8))
axes[0, 0].hist(df_real["Salary"], bins=8, color="teal", edgecolor="black")
axes[0, 0].set_title("Salary Histogram (Numerical)")

df_real["Dept"].value_counts().plot(kind="bar", ax=axes[0, 1], color="salmon")
axes[0, 1].set_title("Department Bar Chart (Categorical)")

axes[1, 0].scatter(df_real["Experience"], df_real["Salary"], color="indigo")
axes[1, 0].set_title("Experience vs Salary Scatter Plot")
axes[1, 0].set_xlabel("Experience (Years)")
axes[1, 0].set_ylabel("Salary ($)")

sns.boxplot(y=df_real["Salary"], ax=axes[1, 1], color="gold")
axes[1, 1].set_title("Salary Outlier Detection Box Plot")
plt.tight_layout()
plt.show()

# 3. Correlation Heatmap
plt.figure(figsize=(5, 4))
sns.heatmap(df_real.select_dtypes(include="number").corr(), annot=True, cmap="Blues")
plt.title("Correlation Heatmap")
plt.show()

# Interpretations:
# 1. Salary distribution shows right skew with high earner outliers.
# 2. Strong positive linear correlation exists between Years of Experience and Salary.
# 3. IT and Sales have the highest department headcount.
