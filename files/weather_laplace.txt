WEATHER PREDICTION USING NAIVE BAYES & LAPLACE ESTIMATOR
================================================================================

PROBLEM STATEMENT:
Predict whether an outdoor tennis match can be played for the test condition:
  Outlook = Overcast, Temperature = Mild, Humidity = High, Wind = Strong
Compare classification results:
  (1) Without Laplace Smoothing (Demonstrating the Zero-Probability problem)
  (2) With Laplace Smoothing (Using additive alpha = 1)

--------------------------------------------------------------------------------
1. PYTHON CODE (COPY & RUN DIRECTLY)
--------------------------------------------------------------------------------
import pandas as pd

# Training Dataset (10 instances)
data = {
    "Outlook":     ["Sunny", "Sunny", "Overcast", "Rain", "Rain", "Rain", "Overcast", "Sunny", "Sunny", "Rain"],
    "Temperature": ["Hot",   "Hot",   "Hot",      "Mild", "Cool", "Cool", "Cool",     "Mild",  "Cool",  "Mild"],
    "Humidity":    ["High",  "High",  "High",     "High", "Normal", "Normal", "Normal", "High", "Normal", "Normal"],
    "Wind":        ["Weak",  "Strong","Weak",     "Weak", "Weak", "Strong", "Strong", "Weak", "Weak",   "Weak"],
    "Play":        ["No",    "No",    "Yes",      "Yes",  "Yes",  "No",     "Yes",    "No",   "Yes",    "Yes"]
}

df = pd.DataFrame(data)

# Query instance
query = {"Outlook": "Overcast", "Temperature": "Mild", "Humidity": "High", "Wind": "Strong"}

df_yes = df[df["Play"] == "Yes"]
df_no  = df[df["Play"] == "No"]

count_yes = len(df_yes)   # 6
count_no  = len(df_no)    # 4
total     = len(df)       # 10

# Priors
P_yes = count_yes / total  # 0.6
P_no  = count_no / total   # 0.4

# Distinct attribute values (k_i)
k_values = {
    "Outlook": 3,      # Sunny, Overcast, Rain
    "Temperature": 3,  # Hot, Mild, Cool
    "Humidity": 2,     # High, Normal
    "Wind": 2          # Weak, Strong
}

score_yes_raw, score_no_raw = P_yes, P_no
score_yes_lap, score_no_lap = P_yes, P_no

print("=== CONDITIONAL PROBABILITIES ===")
for feature, val in query.items():
    c_yes = len(df_yes[df_yes[feature] == val])
    c_no  = len(df_no[df_no[feature] == val])
    k     = k_values[feature]
    
    # Without Laplace
    p_yes_raw = c_yes / count_yes
    p_no_raw  = c_no / count_no
    score_yes_raw *= p_yes_raw
    score_no_raw  *= p_no_raw
    
    # With Laplace: (count + 1) / (N + k)
    p_yes_lap = (c_yes + 1) / (count_yes + k)
    p_no_lap  = (c_no + 1) / (count_no + k)
    score_yes_lap *= p_yes_lap
    score_no_lap  *= p_no_lap
    
    print(f"Feature: {feature} = '{val}'")
    print(f"   P({val}|Yes): Raw = {c_yes}/{count_yes} ({p_yes_raw:.3f}) | Laplace = ({c_yes}+1)/({count_yes}+{k}) = {p_yes_lap:.3f}")
    print(f"   P({val}|No) : Raw = {c_no}/{count_no}  ({p_no_raw:.3f}) | Laplace = ({c_no}+1)/({count_no}+{k}) = {p_no_lap:.3f}")

# Normalize probabilities
norm_yes_raw = score_yes_raw / (score_yes_raw + score_no_raw) if (score_yes_raw + score_no_raw) > 0 else 0
norm_no_raw  = 1 - norm_yes_raw

norm_yes_lap = score_yes_lap / (score_yes_lap + score_no_lap)
norm_no_lap  = score_no_lap / (score_yes_lap + score_no_lap)

print("\n=== FINAL CLASSIFICATION COMPARISON ===")
print(f"WITHOUT LAPLACE:")
print(f"   Score(Yes) = {score_yes_raw:.5f}  |  Score(No) = {score_no_raw:.5f}  (Zero-Probability!)")
print(f"   Prediction: {'Yes (Play)' if score_yes_raw > score_no_raw else 'No (Don\'t Play)'}")

print(f"\nWITH LAPLACE SMOOTHING:")
print(f"   Score(Yes) = {score_yes_lap:.5f}  (Normalized: {norm_yes_lap * 100:.1f}%)")
print(f"   Score(No)  = {score_no_lap:.5f}  (Normalized: {norm_no_lap * 100:.1f}%)")
print(f"   Prediction: {'Yes (Play)' if score_yes_lap > score_no_lap else 'No (Don\'t Play)'}")
