================================================================================
WEATHER PREDICTION USING NAIVE BAYES & LAPLACE ESTIMATOR
================================================================================
Problem Statement:
Predict whether an outdoor game can be conducted for test condition:
    Outlook = Overcast, Temperature = Mild, Humidity = High, Wind = Strong
Compare Naive Bayes probabilities with and without Laplace smoothing.
================================================================================

import pandas as pd

# 1. Create and display the Weather dataset
data = {
    "Outlook":     ["Sunny", "Sunny", "Overcast", "Rain", "Rain", "Rain", "Overcast", "Sunny", "Sunny", "Rain"],
    "Temperature": ["Hot",   "Hot",   "Hot",      "Mild", "Cool", "Cool", "Cool",     "Mild",  "Cool",  "Mild"],
    "Humidity":    ["High",  "High",  "High",     "High", "Normal", "Normal", "Normal", "High", "Normal", "Normal"],
    "Wind":        ["Weak",  "Strong","Weak",     "Weak", "Weak", "Strong", "Strong", "Weak", "Weak",   "Weak"],
    "Play":        ["No",    "No",    "Yes",      "Yes",  "Yes",  "No",     "Yes",    "No",   "Yes",    "Yes"]
}

df = pd.DataFrame(data)
print("=== Weather Dataset ===")
print(df)

# Query instance to classify:
query = {"Outlook": "Overcast", "Temperature": "Mild", "Humidity": "High", "Wind": "Strong"}
print(f"\nQuery Condition to Predict: {query}")

# 2. Separate records according to classes
df_yes = df[df["Play"] == "Yes"]
df_no  = df[df["Play"] == "No"]

total_records = len(df)
count_yes = len(df_yes)
count_no  = len(df_no)

# 3. Calculate Prior Probabilities
P_yes = count_yes / total_records
P_no  = count_no / total_records
print(f"\nPrior Probabilities: P(Yes) = {count_yes}/{total_records} = {P_yes:.4f}, P(No) = {count_no}/{total_records} = {P_no:.4f}")

# Number of distinct categories per attribute (k_i for Laplace)
k_values = {
    "Outlook": len(df["Outlook"].unique()),         # Sunny, Overcast, Rain -> 3
    "Temperature": len(df["Temperature"].unique()), # Hot, Mild, Cool -> 3
    "Humidity": len(df["Humidity"].unique()),       # High, Normal -> 2
    "Wind": len(df["Wind"].unique())               # Weak, Strong -> 2
}

# 4 & 5. Conditional Probabilities: WITHOUT and WITH Laplace Smoothing
print("\n=== Probability Breakdown (Without vs With Laplace Smoothing) ===")

prob_without_laplace = {"Yes": {}, "No": {}}
prob_with_laplace    = {"Yes": {}, "No": {}}

zero_prob_detected = False

for feature, val in query.items():
    # Counts in class Yes
    c_yes = len(df_yes[df_yes[feature] == val])
    p_yes_raw = c_yes / count_yes
    p_yes_lap = (c_yes + 1) / (count_yes + k_values[feature])
    
    # Counts in class No
    c_no = len(df_no[df_no[feature] == val])
    p_no_raw = c_no / count_no
    p_no_lap = (c_no + 1) / (count_no + k_values[feature])
    
    if p_yes_raw == 0 or p_no_raw == 0:
        zero_prob_detected = True
        
    prob_without_laplace["Yes"][feature] = p_yes_raw
    prob_without_laplace["No"][feature]  = p_no_raw
    
    prob_with_laplace["Yes"][feature] = p_yes_lap
    prob_with_laplace["No"][feature]  = p_no_lap
    
    print(f"Feature: {feature} = '{val}' (k={k_values[feature]})")
    print(f"   P({val}|Yes): Raw = {c_yes}/{count_yes} = {p_yes_raw:.4f}  |  Laplace = ({c_yes}+1)/({count_yes}+{k_values[feature]}) = {p_yes_lap:.4f}")
    print(f"   P({val}|No) : Raw = {c_no}/{count_no}  = {p_no_raw:.4f}  |  Laplace = ({c_no}+1)/({count_no}+{k_values[feature]}) = {p_no_lap:.4f}")

# 6. Zero Probability Observation
if zero_prob_detected:
    print("\n[!] Zero-Probability Problem Detected:")
    print("    In the training data, 'Outlook = Overcast' NEVER occurred when Play = No (Count = 0).")
    print("    Without Laplace estimation, P(Overcast | No) = 0, which zeroes out the entire score!")

# 7. Compute Naive Bayes Final Scores
# Score WITHOUT Laplace:
score_yes_raw = P_yes
score_no_raw  = P_no
for f in query:
    score_yes_raw *= prob_without_laplace["Yes"][f]
    score_no_raw  *= prob_without_laplace["No"][f]

# Score WITH Laplace:
score_yes_lap = P_yes
score_no_lap  = P_no
for f in query:
    score_yes_lap *= prob_with_laplace["Yes"][f]
    score_no_lap  *= prob_with_laplace["No"][f]

print("\n=== Final Naive Bayes Scores & Prediction ===")
print("WITHOUT Laplace Smoothing:")
print(f"   Score(Yes) = {score_yes_raw:.6f}")
print(f"   Score(No)  = {score_no_raw:.6f}  (Completely collapsed to 0 due to missing observation!)")
print(f"   Prediction = {'Play = Yes' if score_yes_raw > score_no_raw else 'Play = No'}")

print("\nWITH Laplace Smoothing:")
print(f"   Score(Yes) = {score_yes_lap:.6f}")
print(f"   Score(No)  = {score_no_lap:.6f}")
prediction = "Play = Yes" if score_yes_lap > score_no_lap else "Play = No"
print(f"   Final Prediction = {prediction}")

# 8. Short Observation
print("\n=== Observation on Laplace Estimator ===")
print("In Naive Bayes, all conditional probabilities are multiplied together.")
print("If a particular attribute value (e.g. Outlook=Overcast) never occurs with a class (Play=No),")
print("its probability P(Overcast|No) is 0. Multiplying by 0 wipes out all other evidence.")
print("Laplace estimation adds 1 to the numerator and k to the denominator, guaranteeing non-zero")
print("probabilities for all unseen attribute values while preserving relative rankings.")
