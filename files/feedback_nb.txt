# 5. MULTINOMIAL NAIVE BAYES FROM SCRATCH
# ============================================================
class MultinomialNaiveBayes:
    def __init__(self, alpha):
        self.alpha = alpha
        self.classes = None
        self.class_prior = {}
        self.word_counts = {}
        self.total_words = {}
        self.word_probabilities = {}
        self.vocabulary_size = 0

    def fit(self, X, y):
        self.classes = np.unique(y)
        self.vocabulary_size = X.shape[1]
        total_documents = len(y)

        for current_class in self.classes:
            class_documents = X[y == current_class]
            class_document_count = len(class_documents)
            prior = class_document_count / total_documents
            self.class_prior[current_class] = prior

            counts = np.sum(class_documents, axis=0)
            self.word_counts[current_class] = counts
            total_words = np.sum(counts)
            self.total_words[current_class] = total_words

            # Multinomial Naive Bayes formula with Laplace smoothing:
            # P(w|C) = (Count(w,C) + alpha) / (N_C + alpha * V)
            probabilities = (counts + self.alpha) / (total_words + self.alpha * self.vocabulary_size)
            self.word_probabilities[current_class] = probabilities

        return self

    def predict_one(self, message_vector):
        class_scores = {}
        for current_class in self.classes:
            score = np.log(self.class_prior[current_class])
            log_word_probabilities = np.log(self.word_probabilities[current_class])
            score += np.sum(message_vector * log_word_probabilities)
            class_scores[current_class] = score
        return max(class_scores, key=class_scores.get)

    def predict(self, X):
        return np.array([self.predict_one(message) for message in X])

    def predict_proba(self, X):
        all_probabilities = []
        for message in X:
            log_scores = []
            for current_class in self.classes:
                log_prior = np.log(self.class_prior[current_class])
                log_word_probability = np.log(self.word_probabilities[current_class])
                score = log_prior + np.sum(message * log_word_probability)
                log_scores.append(score)
            log_scores = np.array(log_scores)
            max_score = np.max(log_scores)
            exp_scores = np.exp(log_scores - max_score)
            probabilities = exp_scores / np.sum(exp_scores)
            all_probabilities.append(probabilities)
        return np.array(all_probabilities)

print("""
Multinomial Naive Bayes calculates the probability of a word given a class:
Without smoothing: P(w|C) = Count(w,C) / N_C

If a word has never appeared in a class: Count(w,C) = 0, so P(w|C) = 0.
Because Naive Bayes multiplies word probabilities together, a single zero probability
collapses the entire message score to zero!

Laplace estimation solves this problem by adding alpha=1 to the numerator and V to denominator:
P(w|C) = (Count(w,C) + 1) / (N_C + V)

For an unseen word: Count(w,C) = 0 -> P(w|C) = 1 / (N_C + V) > 0.
This completely prevents the zero-probability problem and allows robust classification.
""")
