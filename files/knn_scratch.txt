def handle_missing_values(data, numeric_method="mean"):
    result = data.copy()
    for column in numeric_columns:
        if result[column].isnull().sum() > 0:
            value = result[column].mean() if numeric_method == "mean" else result[column].median()
            result[column] = result[column].fillna(value)

            mode_value = result[column].mode()[0]

def detect_outliers_iqr(data):
    outlier_counts = {}
    for column in numeric_columns:
        Q1 = data[column].quantile(0.25)
        Q3 = data[column].quantile(0.75)
        IQR = Q3 - Q1
        lower_bound = Q1 - 1.5 * IQR
        upper_bound = Q3 + 1.5 * IQR
        outliers = data[(data[column] < lower_bound) | (data[column] > upper_bound)]
        outlier_counts[column] = len(outliers)
    return outlier_counts

def train_test_split_manual(X, y, test_size=0.2, seed=42):
    rng = np.random.default_rng(seed)

def min_max_fit_transform(X_train, X_test):
    minimum = X_train.min(axis=0)
    maximum = X_train.max(axis=0)
    denominator = maximum - minimum
    denominator[denominator == 0] = 1
    return (X_train - minimum) / denominator, (X_test - minimum) / denominator

def standardize_fit_transform(X_train, X_test):
    mean = X_train.mean(axis=0)
    std = X_train.std(axis=0)
    std[std == 0] = 1
    return (X_train - mean) / std, (X_test - mean) / std

def euclidean_distance(a, b):
    return np.sqrt(np.sum((a - b) ** 2))

def manhattan_distance(a, b):
    return np.sum(np.abs(a - b))

def minkowski_distance(a, b, p=3):
    return (np.sum(np.abs(a - b) ** p)) ** (1 / p)

def cosine_distance(a, b):
    dot = np.dot(a, b)
    norm = np.linalg.norm(a) * np.linalg.norm(b)
    return 1.0 - (dot / norm) if norm != 0 else 1.0

def chebyshev_distance(a, b):
    return np.max(np.abs(a - b))

def calculate_distance(a, b, method="euclidean"):
    if method == "euclidean": return euclidean_distance(a, b)
    elif method == "manhattan": return manhattan_distance(a, b)
    elif method == "minkowski": return minkowski_distance(a, b)
    elif method == "cosine": return cosine_distance(a, b)
    elif method == "chebyshev": return chebyshev_distance(a, b)
    raise ValueError("Unknown distance method")

def knn_predict(X_tr, y_tr, X_te, k, method="euclidean"):
    k_actual = min(k, len(X_tr))
    predictions = []
    for test_pt in X_te:
        distances = [(calculate_distance(test_pt, tr_pt, method), label) for tr_pt, label in zip(X_tr, y_tr)]
        distances.sort(key=lambda x: x[0])
        neighbours = distances[:k_actual]
        labels, counts = np.unique([l for _, l in neighbours], return_counts=True)
        predictions.append(labels[np.argmax(counts)])
    return np.array(predictions)

def weighted_knn_predict(X_tr, y_tr, X_te, k, method="euclidean"):
    k_actual = min(k, len(X_tr))
    predictions = []
    for test_pt in X_te:
        distances = [(calculate_distance(test_pt, tr_pt, method), label) for tr_pt, label in zip(X_tr, y_tr)]
        distances.sort(key=lambda x: x[0])
        neighbours = distances[:k_actual]
        weights = {}
        for d, l in neighbours:
            weights[l] = weights.get(l, 0) + (1.0 / (d + EPSILON))
        predictions.append(max(weights, key=weights.get))
    return np.array(predictions)

def accuracy_score_manual(y_true, y_pred):
    return np.sum(y_true == y_pred) / len(y_true)

def precision_recall_f1(y_true, y_pred):
    labels = np.unique(np.concatenate((y_true, y_pred)))
    precisions, recalls, f1s = [], [], []
    for label in labels:
        TP = np.sum((y_true == label) & (y_pred == label))
        FP = np.sum((y_true != label) & (y_pred == label))
        FN = np.sum((y_true == label) & (y_pred != label))
        p = TP / (TP + FP) if (TP + FP) > 0 else 0
        r = TP / (TP + FN) if (TP + FN) > 0 else 0
        f = (2 * p * r / (p + r)) if (p + r) > 0 else 0
        precisions.append(p); recalls.append(r); f1s.append(f)
    return np.mean(precisions), np.mean(recalls), np.mean(f1s)

def confusion_matrix_manual(y_true, y_pred):
    labels = np.unique(np.concatenate((y_true, y_pred)))
    matrix = np.zeros((len(labels), len(labels)), dtype=int)
    for actual, predicted in zip(y_true, y_pred):
        a_idx = np.where(labels == actual)[0][0]
        p_idx = np.where(labels == predicted)[0][0]
        matrix[a_idx, p_idx] += 1
    return labels, matrix
