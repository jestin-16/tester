PROBLEM STATEMENT:
A healthcare organization has collected medical information from patients to predict
whether a person is at High Risk or Low Risk for cardiovascular disease. Develop a
complete machine learning pipeline using the k-Nearest Neighbour (k-NN) algorithm
implemented entirely from scratch (WITHOUT using sklearn's KNeighborsClassifier).
Only NumPy, Pandas, Matplotlib, and Seaborn may be used.

  - Gender: Male / Female
  - Height: cm

  - BloodPressure: mmHg (systolic)

  - FamilyHistory: Yes / No

  - RiskClass: High Risk / Low Risk (Target)

PIPELINE MODULES IMPLEMENTED FROM SCRATCH:
  - Part A: Data Preparation (Missing value handling, Outlier removal via IQR, Encoding, Min-Max & Z-Score Scaling)
  - Part B: 5 Distance Functions (Euclidean, Manhattan, Minkowski, Cosine, Chebyshev)
  - Part C: Standard Majority-Vote k-NN Classifier
  - Part D: Distance-Weighted k-NN (Weight = 1 / (Distance + 0.0001))
  - Part E: Evaluation across k = 1, 3, 5, 7, 9

import io
import math
import numpy as np
import pandas as pd

# ==============================================================================
# PART A: LOAD 25-ROW CSV DATASET & DATA PREPARATION FROM SCRATCH
# ==============================================================================
csv_data = """PatientID,Age,Gender,Height,Weight,BloodPressure,Glucose,Cholesterol,Smoking,Alcohol,ExerciseLevel,FamilyHistory,HeartRate,RiskClass
P01,63,Male,168,82,145,140,260,Yes,Yes,Low,Yes,88,High Risk
P02,37,Female,160,58,115,88,180,No,No,High,No,68,Low Risk
P03,41,Female,165,64,120,92,195,No,No,Medium,No,72,Low Risk
P04,56,Male,172,88,150,155,275,Yes,No,Low,Yes,82,High Risk

P25,31,Male,182,76,120,94,188,No,No,High,No,69,Low Risk
"""

df = pd.read_csv(io.StringIO(csv_data))
with open("patients.csv", "w") as f:
    f.write(csv_data.strip())

# Compute Body Mass Index (BMI = Weight / Height_m^2)
df["BMI"] = (df["Weight"] / ((df["Height"] / 100.0) ** 2)).round(1)

# 1. Missing Value Imputation (Median for numeric, Mode for categorical)
for num_col in ["BloodPressure", "Glucose", "Cholesterol"]:
    df[num_col] = df[num_col].fillna(df[num_col].median())
df["ExerciseLevel"] = df["ExerciseLevel"].fillna(df["ExerciseLevel"].mode()[0])

# 2. Outlier Detection & Handling via IQR method
for col in ["BloodPressure", "Cholesterol"]:
    Q1 = df[col].quantile(0.25)
    Q3 = df[col].quantile(0.75)
    IQR = Q3 - Q1
    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR
    df = df[(df[col] >= lower_bound) & (df[col] <= upper_bound)]

df = df.reset_index(drop=True)

# 3. Label Encoding from scratch
df["Gender"] = df["Gender"].map({"Male": 1, "Female": 0})
df["Smoking"] = df["Smoking"].map({"Yes": 1, "No": 0})
df["Alcohol"] = df["Alcohol"].map({"Yes": 1, "No": 0})
df["FamilyHistory"] = df["FamilyHistory"].map({"Yes": 1, "No": 0})
df["ExerciseLevel"] = df["ExerciseLevel"].map({"Low": 0, "Medium": 1, "High": 2})
df["Target"] = df["RiskClass"].map({"Low Risk": 0, "High Risk": 1})

# 4. Feature Selection & Min-Max Normalization from scratch
feature_cols = ["Age", "Gender", "Height", "Weight", "BMI", "BloodPressure", "Glucose", "Cholesterol", "Smoking", "Alcohol", "ExerciseLevel", "FamilyHistory", "HeartRate"]
X = df[feature_cols].values
y = df["Target"].values

# Formula: X_norm = (X - X_min) / (X_max - X_min)
X_norm = (X - X.min(axis=0)) / (X.max(axis=0) - X.min(axis=0) + 1e-9)

# Split 80% Train (20 patients), 20% Test (5 patients)
train_size = int(0.80 * len(X_norm))
X_train, X_test = X_norm[:train_size], X_norm[train_size:]
y_train, y_test = y[:train_size], y[train_size:]

# ==============================================================================
# PART B: 5 DISTANCE FUNCTIONS IMPLEMENTED FROM SCRATCH
# ==============================================================================
def euclidean_distance(a, b):
    return math.sqrt(sum((x - y) ** 2 for x, y in zip(a, b)))

def manhattan_distance(a, b):
    return sum(abs(x - y) for x, y in zip(a, b))

def minkowski_distance(a, b, p=3):
    return sum(abs(x - y) ** p for x, y in zip(a, b)) ** (1.0 / p)

def chebyshev_distance(a, b):
    return max(abs(x - y) for x, y in zip(a, b))

def cosine_distance(a, b):
    dot = sum(x * y for x, y in zip(a, b))
    norm_a = math.sqrt(sum(x ** 2 for x in a))
    norm_b = math.sqrt(sum(y ** 2 for y in b))
    return 1.0 - (dot / (norm_a * norm_b)) if (norm_a * norm_b) != 0 else 1.0

# Print distance metrics from Test Patient 0 to first 3 training patients
print("=== PART B: DISTANCE METRICS TO FIRST 3 TRAINING PATIENTS ===")
for i in range(3):
    d_euc = euclidean_distance(X_test[0], X_train[i])
    d_man = manhattan_distance(X_test[0], X_train[i])
    d_cos = cosine_distance(X_test[0], X_train[i])
    print(f"Train Patient P{i+1:02d} | Euclidean: {d_euc:.4f} | Manhattan: {d_man:.4f} | Cosine: {d_cos:.4f}")

# ==============================================================================
# PART C & D: STANDARD & WEIGHTED k-NN PREDICTION FROM SCRATCH
# ==============================================================================
def knn_predict(X_tr, y_tr, query, k=5, weighted=False, epsilon=0.0001):
    distances = [euclidean_distance(query, pt) for pt in X_tr]
    k_indices = np.argsort(distances)[:k]
    k_labels = [y_tr[i] for i in k_indices]
    
    if not weighted:
        # Part C: Standard Majority Vote
        return 1 if sum(k_labels) > (k / 2.0) else 0
    else:
        # Part D: Distance-Weighted Vote (Weight = 1 / (Distance + epsilon))
        k_weights = [1.0 / (distances[i] + epsilon) for i in k_indices]
        w1 = sum(w for l, w in zip(k_labels, k_weights) if l == 1)
        w0 = sum(w for l, w in zip(k_labels, k_weights) if l == 0)
        return 1 if w1 >= w0 else 0

# ==============================================================================
# PART E: EVALUATION ACROSS k = 1, 3, 5, 7, 9
# ==============================================================================
print("\n=== PART E: MODEL ACCURACY ACROSS DIFFERENT k VALUES ===")
k_list = [1, 3, 5, 7, 9]
best_k, best_acc = 1, 0.0

for k in k_list:
    pred_std = [knn_predict(X_train, y_train, pt, k=k, weighted=False) for pt in X_test]
    pred_wgt = [knn_predict(X_train, y_train, pt, k=k, weighted=True) for pt in X_test]
    
    acc_std = np.mean(pred_std == y_test) * 100
    acc_wgt = np.mean(pred_wgt == y_test) * 100
    
    if acc_wgt > best_acc:
        best_acc = acc_wgt
        best_k = k
        
    print(f"k = {k:02d} | Standard k-NN Accuracy: {acc_std:5.1f}% | Weighted k-NN Accuracy: {acc_wgt:5.1f}%")

print(f"\n[+] Optimal Parameter: Best k = {best_k} with {best_acc:.1f}% accuracy.")

# Predict for a new test patient
new_query = X_test[0]
predicted_label = "High Risk" if knn_predict(X_train, y_train, new_query, k=best_k, weighted=True) == 1 else "Low Risk"
actual_label = "High Risk" if y_test[0] == 1 else "Low Risk"
print(f"[+] Prediction on Test Patient 1: Predicted = {predicted_label} | Actual = {actual_label}")
