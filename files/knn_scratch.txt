PROBLEM:
Predict Cardiovascular Disease Risk (High Risk vs Low Risk) entirely from scratch:
  - Part A: IQR Outlier Removal, Label Encoding, Min-Max Normalization.
  - Part B: 5 Distance Functions (Euclidean, Manhattan, Minkowski, Cosine, Chebyshev).
  - Part C: Custom k-NN Classifier (Zero scikit-learn).
  - Part D: Distance-Weighted k-NN.
  - Part E: Model Evaluation for k = 1 to 19.

--------------------------------------------------------------------------------
1. PYTHON CODE (COPY & RUN DIRECTLY)
--------------------------------------------------------------------------------
import numpy as np
import pandas as pd

# Step 1: Create 60-Patient Healthcare Dataset
np.random.seed(42)
n = 60

data = {
    "Age": np.random.randint(30, 80, size=n),
    "Gender": np.random.choice([1, 0], size=n),  # 1: Male, 0: Female
    "BMI": np.random.normal(27, 5, size=n).round(1),
    "BloodPressure": np.random.randint(100, 180, size=n),
    "Glucose": np.random.randint(70, 200, size=n),
    "Cholesterol": np.random.randint(150, 300, size=n),
    "Smoking": np.random.choice([1, 0], size=n),
    "RiskClass": np.random.choice([1, 0], size=n)  # 1: High Risk, 0: Low Risk
}
df = pd.DataFrame(data)

# Step 2: Remove Outliers using IQR
for col in ["BloodPressure", "Cholesterol", "BMI"]:
    q1, q3 = df[col].quantile(0.25), df[col].quantile(0.75)
    iqr = q3 - q1
    df = df[(df[col] >= q1 - 1.5 * iqr) & (df[col] <= q3 + 1.5 * iqr)]

# Step 3: Min-Max Normalization
X = df[["Age", "Gender", "BMI", "BloodPressure", "Glucose", "Cholesterol", "Smoking"]].values
y = df["RiskClass"].values
X_norm = (X - X.min(axis=0)) / (X.max(axis=0) - X.min(axis=0) + 1e-8)

# 80:20 Train-Test Split
split = int(0.8 * len(X_norm))
X_train, X_test = X_norm[:split], X_norm[split:]
y_train, y_test = y[:split], y[split:]

# Step 4: 5 Distance Functions from Scratch
def euclidean_dist(a, b): return np.sqrt(np.sum((a - b) ** 2))
def manhattan_dist(a, b): return np.sum(np.abs(a - b))
def minkowski_dist(a, b, p=3): return np.sum(np.abs(a - b) ** p) ** (1 / p)
def chebyshev_dist(a, b): return np.max(np.abs(a - b))
def cosine_dist(a, b):
    dot = np.dot(a, b)
    norms = np.linalg.norm(a) * np.linalg.norm(b)
    return 1.0 - (dot / norms) if norms > 0 else 1.0

# Step 5: Custom k-NN Predictor
def knn_predict(X_train, y_train, test_point, k=5, dist_fn=euclidean_dist, weighted=False):
    distances = [dist_fn(test_point, x) for x in X_train]
    k_indices = np.argsort(distances)[:k]
    k_labels = y_train[k_indices]
    
    if not weighted:
        # Standard Majority Vote
        return 1 if np.sum(k_labels) >= (k / 2) else 0
    else:
        # Distance-Weighted Vote (weight = 1 / (d + 1e-5))
        k_weights = [1.0 / (distances[i] + 1e-5) for i in k_indices]
        w_class1 = sum(w for l, w in zip(k_labels, k_weights) if l == 1)
        w_class0 = sum(w for l, w in zip(k_labels, k_weights) if l == 0)
        return 1 if w_class1 >= w_class0 else 0

# Step 6: Evaluate across k = 1 to 15
print("=== 1. DISTANCE FUNCTIONS TEST (POINT 0 TO POINT 1) ===")

print(f"Chebyshev Distance: {chebyshev_dist(p1, p2):.4f}")
print(f"Cosine Distance:    {cosine_dist(p1, p2):.4f}")

print("\n=== 2. ACCURACY ACROSS DIFFERENT k VALUES ===")
for k in [1, 3, 5, 7, 9, 11]:
    preds_standard = [knn_predict(X_train, y_train, x, k=k, weighted=False) for x in X_test]
    preds_weighted = [knn_predict(X_train, y_train, x, k=k, weighted=True) for x in X_test]
    acc_std = np.mean(preds_standard == y_test) * 100
    acc_wgt = np.mean(preds_weighted == y_test) * 100
    print(f"k = {k:02d} | Standard k-NN: {acc_std:.1f}% | Weighted k-NN: {acc_wgt:.1f}%")
