Perform 12 specific operations on a 30+ record CSV dataset:

import numpy as np
import pandas as pd

# Step 0: Create sample raw dataset with 32 records (including duplicates & missing values)
np.random.seed(42)
raw_data = {
    "StudentID": [f"S{i:03d}" for i in range(1, 31)] + ["S005", "S012"],  # 2 duplicates
    "Name": [f"Student_{i}" for i in range(1, 31)] + ["Student_5", "Student_12"],
    "Department": np.random.choice(["CSE", "ECE", "IT", "MECH"], size=32),
    "Maths": np.random.randint(40, 100, size=32).astype(float),
    "Science": np.random.randint(40, 100, size=32).astype(float),
    "English": np.random.randint(40, 100, size=32).astype(float)
}

# Inject missing values for demonstration
raw_data["Maths"][2] = np.nan
raw_data["Science"][7] = np.nan
raw_data["Department"][15] = None

pd.DataFrame(raw_data).to_csv("Raw_Students.csv", index=False)
print("[+] Created 'Raw_Students.csv' for demonstration.\n")

# ------------------------------------------------------------------------------
# 1. Load dataset, display dimensions, column names, and data types
# ------------------------------------------------------------------------------
df = pd.read_csv("Raw_Students.csv")
print("1. Dataset Dimensions:", df.shape)
print("   Column Names:", list(df.columns))
print("   Data Types:\n", df.dtypes)

# ------------------------------------------------------------------------------
# 2. Display first 5 and last 5 records
# ------------------------------------------------------------------------------
print("\n2. First 5 records:\n", df.head())
print("\n   Last 5 records:\n", df.tail())

# ------------------------------------------------------------------------------
# 3. Check and remove duplicate records
# ------------------------------------------------------------------------------
num_duplicates = df.duplicated().sum()
print(f"\n3. Duplicate records found: {num_duplicates}")
df = df.drop_duplicates()
print(f"   Updated number of records: {len(df)}")

# ------------------------------------------------------------------------------
# 4. Identify & handle missing values (Mean for numerical, Mode for categorical)
# ------------------------------------------------------------------------------
print("\n4. Missing values per column:\n", df.isnull().sum())
# Fill numerical columns with mean
num_cols = df.select_dtypes(include=[np.number]).columns
for col in num_cols:
    df[col] = df[col].fillna(df[col].mean())

# Fill categorical columns with mode
cat_cols = df.select_dtypes(include=["object"]).columns
for col in cat_cols:
    df[col] = df[col].fillna(df[col].mode()[0])

print("   Missing values after imputation:\n", df.isnull().sum())

# ------------------------------------------------------------------------------
# 5. Statistical summary of numerical columns
# ------------------------------------------------------------------------------
print("\n5. Statistical Summary of Numerical Columns:\n", df.describe())

# ------------------------------------------------------------------------------
# 6. Create a new calculated column (Total Marks & Average Marks)
# ------------------------------------------------------------------------------
df["Total_Marks"] = df["Maths"] + df["Science"] + df["English"]
df["Average_Marks"] = df["Total_Marks"] / 3
print("\n6. Added calculated column 'Total_Marks' and 'Average_Marks'.")

# ------------------------------------------------------------------------------
# 7. Add classification column based on conditions (Grade)
# ------------------------------------------------------------------------------
def assign_grade(avg):
    if avg >= 85:
        return "Grade A"
    elif avg >= 70:
        return "Grade B"
    elif avg >= 50:
        return "Grade C"
    else:
        return "Fail"

df["Grade"] = df["Average_Marks"].apply(assign_grade)
print("\n7. Grade Classification distribution:\n", df["Grade"].value_counts())

# ------------------------------------------------------------------------------
# 8. Top 5 and Bottom 5 records based on Total_Marks
# ------------------------------------------------------------------------------
print("\n8. Top 5 Records by Total Marks:\n", df.sort_values(by="Total_Marks", ascending=False).head(5))
print("\n   Bottom 5 Records by Total Marks:\n", df.sort_values(by="Total_Marks", ascending=True).head(5))

# ------------------------------------------------------------------------------
# 9. Statistical properties of the main numerical column (Total_Marks)
# ------------------------------------------------------------------------------
print("\n9. Statistics for Total_Marks:")
print(f"   Mean:               {df['Total_Marks'].mean():.2f}")
print(f"   Median:             {df['Total_Marks'].median():.2f}")
print(f"   Mode:               {df['Total_Marks'].mode().iloc[0]:.2f}")
print(f"   Standard Deviation: {df['Total_Marks'].std():.2f}")
print(f"   Variance:           {df['Total_Marks'].var():.2f}")

# ------------------------------------------------------------------------------
# 10. Group by categorical column (Department) and aggregate
# ------------------------------------------------------------------------------
dept_grp = df.groupby("Department")["Total_Marks"].agg(["mean", "max", "min"]).reset_index()
print("\n10. Department-wise Marks Summary:\n", dept_grp)

# ------------------------------------------------------------------------------
# 11. Filter records where numerical value > overall average
# ------------------------------------------------------------------------------
overall_avg = df["Total_Marks"].mean()
above_avg = df[df["Total_Marks"] > overall_avg]
print(f"\n11. Records scoring above overall average ({overall_avg:.2f}): {len(above_avg)} students found.")

# ------------------------------------------------------------------------------
# 12. Rank records and display Top 3
# ------------------------------------------------------------------------------
df["Rank"] = df["Total_Marks"].rank(ascending=False, method="dense").astype(int)
print("\n12. Top 3 Ranked Students:\n", df.sort_values(by="Rank").head(3)[["Rank", "StudentID", "Name", "Total_Marks", "Grade"]])

# ------------------------------------------------------------------------------
# 13. Export processed dataset to CSV
# ------------------------------------------------------------------------------
df.to_csv("Processed_Report.csv", index=False)
print("\n[+] Successfully exported to 'Processed_Report.csv'!")
