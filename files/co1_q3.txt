PROBLEM STATEMENT:
Download or create a CSV dataset containing at least 30 records.
Write a Python program using the Pandas library to perform the following tasks:
  1. Load the dataset and display its dimensions, column names, and data types.
  2. Display the first five and last five records.
  3. Check whether the dataset contains duplicate records. If duplicates exist,
     remove them and display the updated number of records.
  4. Identify missing values in each column. Fill missing numerical values with
     the column mean and missing categorical values with the mode.
  5. Generate the statistical summary of the numerical columns.
  6. Create a new calculated column based on the dataset (Total Marks, Average).
  7. Add another column that classifies each record into categories (Grade).
  8. Find the top five and bottom five records based on the calculated column.
  9. Calculate the mean, median, mode, standard deviation, and variance of the
     main numerical column.
 10. Group the dataset by one categorical column and calculate the average,
     maximum, and minimum values of another numerical column.
 11. Display only those records whose numerical value is greater than the overall average.
 12. Rank the records based on the numerical column and display the top three ranked records.
 13. Export the processed dataset to a new CSV file named Processed_Report.csv.

--------------------------------------------------------------------------------
DATASET: 32 RECORDS WITH INTENTIONAL MISSING VALUES & DUPLICATES
--------------------------------------------------------------------------------
Fields: StudentID, Name, Department, Maths, Science, English

--------------------------------------------------------------------------------
PYTHON SOLUTION (DATASET CREATION & COMPLETE ANSWERS)
--------------------------------------------------------------------------------
import numpy as np
import pandas as pd

# Step 1: Create 32-row Dataset (30 unique + 2 duplicates + intentional missing values)
np.random.seed(42)
departments = ["CSE", "ECE", "MECH", "IT"]

records = []
for i in range(1, 31):
    records.append({
        "StudentID": f"S{i:03d}",
        "Name": f"Student_{i}",
        "Department": np.random.choice(departments),
        "Maths": float(np.random.randint(40, 100)),
        "Science": float(np.random.randint(40, 100)),
        "English": float(np.random.randint(40, 100))
    })

# Add 2 duplicate rows
records.append(records[0].copy())
records.append(records[5].copy())

df_raw = pd.DataFrame(records)

# Introduce realistic missing values (NaN)
df_raw.loc[2, "Maths"] = np.nan
df_raw.loc[7, "Science"] = np.nan
df_raw.loc[12, "Department"] = np.nan
df_raw.to_csv("students_raw.csv", index=False)

# ------------------------------------------------------------------------------
# PERFORM ALL 13 PANDAS OPERATIONS
# ------------------------------------------------------------------------------

# 1. Load dataset & inspect structure
df = pd.read_csv("students_raw.csv")
print("1. Shape (Dimensions):", df.shape)
print("   Column Names:      ", list(df.columns))
print("   Data Types:\n", df.dtypes)

# 2. Display first 5 and last 5 records
print("\n2. First 5 Records:\n", df.head())
print("\n   Last 5 Records:\n", df.tail())

# 3. Check and remove duplicate records
dup_count = df.duplicated().sum()
df = df.drop_duplicates().reset_index(drop=True)
print(f"\n3. Duplicate rows removed: {dup_count}. Updated records: {len(df)}")

# 4. Identify & Impute missing values (Mean for numeric, Mode for categorical)
print("\n4. Missing values before imputation:\n", df.isnull().sum())
for col in ["Maths", "Science", "English"]:
    df[col] = df[col].fillna(df[col].mean().round(2))
df["Department"] = df["Department"].fillna(df["Department"].mode()[0])
print("   Missing values after imputation:\n", df.isnull().sum())

# 5. Statistical Summary of numerical columns
print("\n5. Statistical Summary:\n", df[["Maths", "Science", "English"]].describe().round(2))

# 6. Create New Calculated Columns
df["Total_Marks"] = df["Maths"] + df["Science"] + df["English"]
df["Average_Marks"] = (df["Total_Marks"] / 3).round(2)

# 7. Classification Column (Grade)
def get_grade(avg):
    if avg >= 85: return "Grade A"
    elif avg >= 70: return "Grade B"
    elif avg >= 50: return "Grade C"
    return "Fail"

df["Grade"] = df["Average_Marks"].apply(get_grade)

# 8. Top 5 and Bottom 5 Records
print("\n8. Top 5 Records:\n", df.sort_values(by="Total_Marks", ascending=False).head(5)[["StudentID", "Name", "Total_Marks", "Grade"]].to_string(index=False))
print("\n   Bottom 5 Records:\n", df.sort_values(by="Total_Marks", ascending=True).head(5)[["StudentID", "Name", "Total_Marks", "Grade"]].to_string(index=False))

# 9. Main Numerical Column Statistics (Total_Marks)
total_stats = pd.Series({
    "Mean": df["Total_Marks"].mean().round(2),
    "Median": df["Total_Marks"].median().round(2),
    "Mode": df["Total_Marks"].mode().iloc[0],
    "Std_Dev": df["Total_Marks"].std().round(2),
    "Variance": df["Total_Marks"].var().round(2)
})
print("\n9. Total_Marks Statistics:\n", total_stats)

# 10. Groupby Department and calculate Avg, Max, Min of Total_Marks
dept_summary = df.groupby("Department")["Total_Marks"].agg(["mean", "max", "min"]).round(2)
print("\n10. Department-wise Summary of Total Marks:\n", dept_summary)

# 11. Records greater than overall Average
class_avg = df["Average_Marks"].mean()
above_avg_df = df[df["Average_Marks"] > class_avg]
print(f"\n11. Records with Average > Class Average ({class_avg:.2f}):\n", above_avg_df[["StudentID", "Name", "Department", "Average_Marks"]].head().to_string(index=False))

# 12. Rank records and display Top 3
df["Rank"] = df["Total_Marks"].rank(ascending=False, method="dense").astype(int)
print("\n12. Top 3 Ranked Students:\n", df.sort_values("Rank").head(3)[["Rank", "StudentID", "Name", "Total_Marks", "Grade"]].to_string(index=False))

# 13. Export processed dataset to Processed_Report.csv
df.sort_values(by="Rank").to_csv("Processed_Report.csv", index=False)
print("\n13. Successfully exported processed dataset to 'Processed_Report.csv'.")
