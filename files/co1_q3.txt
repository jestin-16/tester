PROBLEM:
Perform 12 essential data operations on a 30+ record dataset:
  1. Load dataset, display shape, columns, and data types
  2. Display head(5) and tail(5)
  3. Check and drop duplicate records
  4. Impute missing values (Mean for numerical, Mode for categorical)
  5. Statistical summary using describe()
  6. Add calculated columns (Total_Marks and Average_Marks)
  7. Add Grade classification column (A, B, C, Fail)
  8. Display Top 5 and Bottom 5 performers
  9. Groupby Department analysis
  10. Filter students performing above overall class average
  11. Rank students by Total_Marks (dense ranking)
  12. Export final clean dataset to 'Processed_Report.csv'

--------------------------------------------------------------------------------
1. PYTHON CODE (COPY & RUN DIRECTLY)
--------------------------------------------------------------------------------
import numpy as np
import pandas as pd

# Step 0: Generate sample CSV with 32 records (with duplicates & missing values)
np.random.seed(42)
raw_data = {
    "StudentID": [f"S{i:03d}" for i in range(1, 31)] + ["S005", "S012"],
    "Name": [f"Student_{i}" for i in range(1, 31)] + ["Student_5", "Student_12"],
    "Department": np.random.choice(["CSE", "ECE", "IT", "MECH"], size=32),
    "Maths": np.random.randint(40, 100, size=32).astype(float),
    "Science": np.random.randint(40, 100, size=32).astype(float),
    "English": np.random.randint(40, 100, size=32).astype(float)
}
# Inject missing values for demonstration
raw_data["Maths"][2] = np.nan
raw_data["Science"][7] = np.nan
raw_data["Department"][15] = None

pd.DataFrame(raw_data).to_csv("Raw_Students.csv", index=False)

# 1. Load dataset, display dimensions, columns, and dtypes
df = pd.read_csv("Raw_Students.csv")
print("1. Shape:", df.shape)
print("   Columns:", list(df.columns))
print("   Data Types:\n", df.dtypes)

# 2. Display first 5 and last 5 records
print("\n2. Head:\n", df.head(3))
print("   Tail:\n", df.tail(3))

# 3. Check and remove duplicate records
dups = df.duplicated().sum()
print(f"\n3. Duplicate rows removed: {dups}")
df = df.drop_duplicates()

# 4. Handle Missing Values (Mean for numerical, Mode for categorical)
for col in df.select_dtypes(include=[np.number]).columns:
    df[col] = df[col].fillna(df[col].mean().round(2))

for col in df.select_dtypes(include=["object"]).columns:
    df[col] = df[col].fillna(df[col].mode()[0])

print(f"4. Missing values remaining: {df.isnull().sum().sum()}")

# 5. Summary statistics
print("\n5. Summary Statistics:\n", df.describe().round(2))

# 6. Calculated Columns
df["Total_Marks"] = df["Maths"] + df["Science"] + df["English"]
df["Average_Marks"] = (df["Total_Marks"] / 3).round(2)

# 7. Classification Column (Grade)
def get_grade(avg):
    if avg >= 85: return "Grade A"
    elif avg >= 70: return "Grade B"
    elif avg >= 50: return "Grade C"
    return "Fail"

df["Grade"] = df["Average_Marks"].apply(get_grade)

# 8. Top 5 and Bottom 5
print("\n8. Top 3 Performers:\n", df.sort_values(by="Total_Marks", ascending=False)[["StudentID", "Name", "Total_Marks", "Grade"]].head(3).to_string(index=False))

# 9. Groupby Department
print("\n9. Department Averages:\n", df.groupby("Department")[["Total_Marks", "Average_Marks"]].mean().round(2))

# 10. Filter above class average
class_mean = df["Average_Marks"].mean()
above_avg = df[df["Average_Marks"] > class_mean]
print(f"\n10. Students Above Class Average ({class_mean:.2f}): {len(above_avg)} students")

# 11. Rank Students
df["Rank"] = df["Total_Marks"].rank(ascending=False, method="dense").astype(int)

# 12. Export to CSV
df.sort_values(by="Rank").to_csv("Processed_Report.csv", index=False)
print("\n12. Successfully exported clean dataset to 'Processed_Report.csv'.")
