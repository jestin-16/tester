1. CORE PANDAS WORKFLOW:
df = pd.read_csv("dataset.csv")
print(df.shape, df.columns, df.dtypes)
print(df.head(), df.tail(), df.describe(), df.isnull().sum())
df = df.drop_duplicates()

# Imputation
df["NumCol"] = df["NumCol"].fillna(df["NumCol"].mean())
df["CatCol"] = df["CatCol"].fillna(df["CatCol"].mode()[0])

# Feature engineering & Grouping
df["Total"] = df[["M1", "M2", "M3"]].sum(axis=1)
print(df.groupby("Department")["Total"].agg(["mean", "max", "min"]))

# Ranking & Export
df["Rank"] = df["Total"].rank(ascending=False)
df.sort_values("Rank").to_csv("Processed_Report.csv", index=False)
