df = pd.read_csv(io.StringIO(csv_data))
with open("patients.csv", "w") as f:
    f.write(csv_data.strip())

df["BMI"] = (df["Weight"] / ((df["Height"] / 100.0) ** 2)).round(1)

# Imputation
for num_col in ["BloodPressure", "Glucose", "Cholesterol"]:
    df[num_col] = df[num_col].fillna(df[num_col].median())
df["ExerciseLevel"] = df["ExerciseLevel"].fillna(df["ExerciseLevel"].mode()[0])

# Outlier handling via IQR
for col in ["BloodPressure", "Cholesterol"]:
    Q1 = df[col].quantile(0.25)
    Q3 = df[col].quantile(0.75)
    IQR = Q3 - Q1
    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR
    df = df[(df[col] >= lower_bound) & (df[col] <= upper_bound)]

df = df.reset_index(drop=True)

# Label encoding
df["Gender"] = df["Gender"].map({"Male": 1, "Female": 0})
df["Smoking"] = df["Smoking"].map({"Yes": 1, "No": 0})
df["Alcohol"] = df["Alcohol"].map({"Yes": 1, "No": 0})
df["FamilyHistory"] = df["FamilyHistory"].map({"Yes": 1, "No": 0})
df["ExerciseLevel"] = df["ExerciseLevel"].map({"Low": 0, "Medium": 1, "High": 2})
df["Target"] = df["RiskClass"].map({"Low Risk": 0, "High Risk": 1})

# Feature normalization (Min-Max Scaling from scratch)
feature_cols = ["Age", "Gender", "Height", "Weight", "BMI", "BloodPressure", "Glucose", "Cholesterol", "Smoking", "Alcohol", "ExerciseLevel", "FamilyHistory", "HeartRate"]
X = df[feature_cols].values
y = df["Target"].values
X_norm = (X - X.min(axis=0)) / (X.max(axis=0) - X.min(axis=0) + 1e-9)

train_size = int(0.80 * len(X_norm))
X_train, X_test = X_norm[:train_size], X_norm[train_size:]
y_train, y_test = y[:train_size], y[train_size:]

# Part B: 5 Distance Functions From Scratch
def euclidean_distance(a, b):
    return math.sqrt(sum((x - y) ** 2 for x, y in zip(a, b)))

def manhattan_distance(a, b):
    return sum(abs(x - y) for x, y in zip(a, b))

def minkowski_distance(a, b, p=3):
    return sum(abs(x - y) ** p for x, y in zip(a, b)) ** (1.0 / p)

def chebyshev_distance(a, b):
    return max(abs(x - y) for x, y in zip(a, b))

def cosine_distance(a, b):
    dot = sum(x * y for x, y in zip(a, b))
    norm_a = math.sqrt(sum(x ** 2 for x in a))
    norm_b = math.sqrt(sum(y ** 2 for y in b))
    return 1.0 - (dot / (norm_a * norm_b)) if (norm_a * norm_b) != 0 else 1.0

# Part C & D: Majority-Vote & Distance-Weighted k-NN
def knn_predict(X_tr, y_tr, query, k=5, weighted=False, epsilon=0.0001):
    distances = [euclidean_distance(query, pt) for pt in X_tr]
    k_indices = np.argsort(distances)[:k]
    k_labels = [y_tr[i] for i in k_indices]
    
    if not weighted:
        return 1 if sum(k_labels) > (k / 2.0) else 0
    else:
        k_weights = [1.0 / (distances[i] + epsilon) for i in k_indices]
        w1 = sum(w for l, w in zip(k_labels, k_weights) if l == 1)
        w0 = sum(w for l, w in zip(k_labels, k_weights) if l == 0)
        return 1 if w1 >= w0 else 0

# Part E: Evaluation across k values
print("\n=== PART E: MODEL ACCURACY ACROSS DIFFERENT k VALUES ===")
k_list = [1, 3, 5, 7, 9]
best_k, best_acc = 1, 0.0

for k in k_list:
    pred_std = [knn_predict(X_train, y_train, pt, k=k, weighted=False) for pt in X_test]
    pred_wgt = [knn_predict(X_train, y_train, pt, k=k, weighted=True) for pt in X_test]
    
    acc_std = np.mean(pred_std == y_test) * 100
    acc_wgt = np.mean(pred_wgt == y_test) * 100
    
    if acc_wgt > best_acc:
        best_acc = acc_wgt
        best_k = k
        
    print(f"k = {k:02d} | Standard k-NN Accuracy: {acc_std:5.1f}% | Weighted k-NN Accuracy: {acc_wgt:5.1f}%")

print(f"\n[+] Optimal Parameter: Best k = {best_k} with {best_acc:.1f}% accuracy.")

new_query = X_test[0]
predicted_label = "High Risk" if knn_predict(X_train, y_train, new_query, k=best_k, weighted=True) == 1 else "Low Risk"
actual_label = "High Risk" if y_test[0] == 1 else "Low Risk"
print(f"[+] Prediction on Test Patient 1: Predicted = {predicted_label} | Actual = {actual_label}")

# Step 2: Inspect first 5 records & class distribution
print("=== 1. DATASET OVERVIEW & CLASS DISTRIBUTION ===")
print("First 5 records:\n", df.head())
print("\nClass Distribution:\n", df["Class"].value_counts())

# Step 3: Vectorize text using CountVectorizer (Bag of Words)
vectorizer = CountVectorizer(stop_words="english")
X = vectorizer.fit_transform(df["Message"])
y = df["Class"]
vocab = vectorizer.get_feature_names_out()

# Step 4: Split into 80% Train (20 messages), 20% Test (5 messages)
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.20, random_state=42, stratify=y
)

# Step 5: Train Model WITHOUT Laplace Smoothing (alpha = 1e-9)
model_no_lap = MultinomialNB(alpha=1e-9)
model_no_lap.fit(X_train, y_train)

# Step 6: Train Model WITH Laplace Smoothing (alpha = 1.0)
model_with_lap = MultinomialNB(alpha=1.0)
model_with_lap.fit(X_train, y_train)

# Step 7 & 8: Unseen Word Probability Demonstration
V = len(vocab)
classes = model_with_lap.classes_
print("\n=== 2. UNSEEN WORD PROBABILITY FORMULA ===")
print(f"Total Unique Vocabulary Size (V): {V} words")

unseen_word = "quantum"
print(f"For an unseen test word like '{unseen_word}':")
for i, c in enumerate(classes):
    N_c = model_with_lap.feature_count_[i].sum()
    p_unseen = 1.0 / (N_c + V)
    print(f"   In Class '{c}': Count=0, Total Words N_C={int(N_c)}, P('{unseen_word}'|{c}) = (0+1)/({int(N_c)}+{V}) = {p_unseen:.6f}")

# Step 9: Compare Predictions on Test Split
preds_no = model_no_lap.predict(X_test)
preds_lap = model_with_lap.predict(X_test)

comparison_df = pd.DataFrame({
    "Test_Message": [df["Message"].iloc[idx] for idx in y_test.index],
    "Actual_Class": y_test.values,
    "No_Laplace_Prediction": preds_no,
    "With_Laplace_Prediction": preds_lap
})
print("\n=== 3. TEST SET PREDICTIONS COMPARISON ===")
print(comparison_df.to_string(index=False))

# Step 10 & 11: Accept New Feedback & Predict with Probabilities
new_sample = "The teacher explained difficult topics with outstanding clarity"
new_vec = vectorizer.transform([new_sample])

pred_class = model_with_lap.predict(new_vec)[0]
probs = dict(zip(model_with_lap.classes_, model_with_lap.predict_proba(new_vec)[0].round(4)))

print(f"\n=== 4. PREDICTION ON NEW USER FEEDBACK ===")
print(f"Input Message:  '{new_sample}'")
print(f"Predicted Class: {pred_class}")
print(f"Class Probabilities: {probs}")

# Step 12: Explanation of Zero-Probability Problem
print("\n=== 5. HOW LAPLACE ESTIMATION PREVENTS ZERO PROBABILITIES ===")
print("In Naive Bayes, the class score is computed as the product of conditional probabilities:")
print("   P(Class) * P(w1|Class) * P(w2|Class) * ... * P(wn|Class)")
print("If a single word does not appear in a training class, its count is 0, making P(wi|Class) = 0.")
print("This single zero causes the entire multiplicative product to collapse to zero, regardless of other words.")
print("Laplace smoothing adds alpha=1 to the numerator and V to the denominator, guaranteeing every word")
print("has a non-zero probability and ensuring robust classification on real-world text.")
