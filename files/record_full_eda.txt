import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# Step 1: Create 120-Record Public-Style Dataset (Kaggle/UCI Churn Pattern)
np.random.seed(42)
n = 120
contracts = ["Month-to-Month", "One Year", "Two Year"]

data = {
    "CustomerID": [f"CUST_{i:04d}" for i in range(1, n + 1)],
    "Tenure_Months": np.random.randint(1, 72, size=n),
    "Monthly_Charges": np.random.uniform(20.0, 110.0, size=n).round(2),
    "Contract_Type": np.random.choice(contracts, size=n, p=[0.5, 0.3, 0.2]),
    "Customer_Support_Calls": np.random.poisson(lam=2, size=n),
    "Churn_Status": np.random.choice(["No", "Yes"], size=n, p=[0.72, 0.28])
}

df = pd.DataFrame(data)
# Total Spend = Tenure * Monthly Charges with random discount/variance
df["Total_Spend"] = (df["Tenure_Months"] * df["Monthly_Charges"] * np.random.uniform(0.95, 1.05, size=n)).round(2)

# Introduce 2 missing values to demonstrate handling
df.loc[7, "Monthly_Charges"] = np.nan
df.loc[22, "Tenure_Months"] = np.nan

# ------------------------------------------------------------------------------
# 1. DISPLAY DATASET INFORMATION
# ------------------------------------------------------------------------------
print("=== 1. DATASET OVERVIEW ===")
print("Shape:", df.shape)
print("\nData Types & Info:")
print(df.dtypes)
print("\nFirst 5 Records:\n", df.head())

# ------------------------------------------------------------------------------
# 2. HANDLE MISSING VALUES
# ------------------------------------------------------------------------------
print("\nMissing values before handling:\n", df.isnull().sum())
df["Monthly_Charges"] = df["Monthly_Charges"].fillna(df["Monthly_Charges"].mean())
df["Tenure_Months"] = df["Tenure_Months"].fillna(df["Tenure_Months"].median())
df["Total_Spend"] = df["Total_Spend"].fillna(df["Tenure_Months"] * df["Monthly_Charges"])
print("Missing values after handling:\n", df.isnull().sum())

# ------------------------------------------------------------------------------
# 3. DESCRIPTIVE STATISTICS
# ------------------------------------------------------------------------------
print("\n=== 2. DESCRIPTIVE STATISTICS ===")
print(df.describe().round(2))

# ------------------------------------------------------------------------------
# 4. EIGHT CORE VISUALIZATIONS
# ------------------------------------------------------------------------------

# 1. Line Chart: Average Total Spend by Tenure
tenure_spend = df.groupby("Tenure_Months")["Total_Spend"].mean()
plt.figure(figsize=(8, 4))
plt.plot(tenure_spend.index, tenure_spend.values, color="crimson", marker="o", markersize=3)
plt.title("1. Customer Lifetime Value (Spend vs Tenure)")
plt.xlabel("Tenure (Months)")
plt.ylabel("Average Total Spend ($)")
plt.grid(True)
plt.show()

# 2. Bar Chart: Customer Distribution by Contract Type
plt.figure(figsize=(6, 4))
df["Contract_Type"].value_counts().plot(kind="bar", color=["#4c72b0", "#55a868", "#c44e52"], edgecolor="black")
plt.title("2. Customer Count by Contract Type")
plt.xlabel("Contract Type")
plt.ylabel("Count")
plt.grid(axis="y", linestyle=":")
plt.show()

# 3. Histogram: Distribution of Monthly Charges
plt.figure(figsize=(7, 4))
plt.hist(df["Monthly_Charges"], bins=10, color="darkcyan", edgecolor="black")
plt.title("3. Distribution of Monthly Charges ($)")
plt.xlabel("Monthly Charge Range")
plt.ylabel("Customer Count")
plt.grid(axis="y")
plt.show()

# 4. Pie Chart: Overall Churn Ratio
plt.figure(figsize=(5, 5))
churn_counts = df["Churn_Status"].value_counts()
plt.pie(churn_counts, labels=churn_counts.index, autopct="%1.1f%%", colors=["#66b3ff", "#ff9999"], explode=(0, 0.08), shadow=True)
plt.title("4. Churn Distribution (Active vs Churned)")
plt.show()

# 5. Scatter Plot: Monthly Charges vs Total Spend by Churn
plt.figure(figsize=(8, 5))
sns.scatterplot(data=df, x="Monthly_Charges", y="Total_Spend", hue="Churn_Status", style="Contract_Type", s=90, palette="Set1")
plt.title("5. Monthly Charges vs Total Spend (by Churn Status)")
plt.xlabel("Monthly Charges ($)")
plt.ylabel("Total Spend ($)")
plt.grid(True)
plt.show()

# 6. Box Plot: Monthly Charges Across Contract Types (Outlier Detection)
plt.figure(figsize=(7, 4))
sns.boxplot(data=df, x="Contract_Type", y="Monthly_Charges", palette="Pastel1")
plt.title("6. Monthly Charges by Contract Type (Outlier Check)")
plt.show()

# 7. Correlation Heatmap
plt.figure(figsize=(6, 4))
num_df = df.select_dtypes(include="number")
sns.heatmap(num_df.corr(), annot=True, cmap="YlGnBu", fmt=".2f", linewidths=0.5)
plt.title("7. Correlation Matrix of Numerical Features")
plt.show()

# 8. Pair Plot
sns.pairplot(df[["Tenure_Months", "Monthly_Charges", "Total_Spend", "Churn_Status"]], hue="Churn_Status", palette="husl")
plt.show()

# ------------------------------------------------------------------------------
# 8-10 STATISTICAL OBSERVATIONS
# ------------------------------------------------------------------------------
observations = """
